import allel
import pandas as pd
import numpy as np
from itertools import combinations

VCF_FILE = "vcf.gz"
POP_MAP = "pop_map.txt"
OUT_FILE = "Fst_results.csv"

# LOAD POPULATION MAP

pop_df = pd.read_csv(POP_MAP, sep=r"\s+", engine="python")
pop_df.columns = pop_df.columns.str.strip()

assert "SampleID" in pop_df.columns
assert "Population" in pop_df.columns

# LOAD VCF

callset = allel.read_vcf(
    VCF_FILE,
    fields=["samples", "calldata/GT"],
    alt_number=1
)

samples = callset["samples"]
 
# Build GenotypeArray explicitly

gt = allel.GenotypeArray(callset["calldata/GT"])

print(f"Loaded {len(samples)} samples and {gt.n_variants} SNPs")


# BUILD SUBPOPULATION INDICES

sample_to_index = {s: i for i, s in enumerate(samples)}
Filter SNPs (recommended for F_ST)
subpops = {}
for pop in pop_df["Population"].unique():
    inds = [
        sample_to_index[s]
        for s in pop_df.loc[pop_df["Population"] == pop, "SampleID"]
        if s in sample_to_index
    ]
    if len(inds) < 2:
        raise ValueError(f"Population {pop} has <2 samples")
    subpops[pop] = inds
    print(f"{pop}: {len(inds)} samples")


# PAIRWISE HUDSON FST

results = []

for pop1, pop2 in combinations(subpops.keys(), 2):

    inds1 = subpops[pop1]
    inds2 = subpops[pop2]

    # Count alleles PER POPULATION
    ac1 = gt.take(inds1, axis=1).count_alleles()
    ac2 = gt.take(inds2, axis=1).count_alleles()

    # Hudson FST
    num, den = allel.hudson_fst(ac1, ac2)
    fst = np.nansum(num) / np.nansum(den)

    results.append({
        "Population_1": pop1,
        "Population_2": pop2,
        "Hudson_Fst": fst
    })

    print(f"{pop1} vs {pop2}: FST = {fst:.4f}")


# SAVE OUTPUT

df = pd.DataFrame(results)
df.to_csv(OUT_FILE, index=False)

print(f"\nSaved results to {OUT_FILE}")
